# rencm-gui - ImGui front-end for the rencm library.
#
# Owns everything the library does not need: GLFW + Dear ImGui, the bundled
# fonts (gzip-compressed, embedded via .incbin) and the application sources.

# ---------------------------------------------------------------- GLFW
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(glfw
  GIT_REPOSITORY https://github.com/glfw/glfw.git
  GIT_TAG 3.4
  GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(glfw)

# ---------------------------------------------------------------- Dear ImGui
FetchContent_Declare(imgui
  GIT_REPOSITORY https://github.com/ocornut/imgui.git
  GIT_TAG v1.92.9b
  GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(imgui)

add_library(imgui_backend STATIC
  ${imgui_SOURCE_DIR}/imgui.cpp
  ${imgui_SOURCE_DIR}/imgui_draw.cpp
  ${imgui_SOURCE_DIR}/imgui_tables.cpp
  ${imgui_SOURCE_DIR}/imgui_widgets.cpp
  ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp
  ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
target_include_directories(imgui_backend PUBLIC
  ${imgui_SOURCE_DIR}
  ${imgui_SOURCE_DIR}/backends)
target_link_libraries(imgui_backend PUBLIC glfw)

# ---------------------------------------------------------------- miniaudio
# Single-header, public-domain playback engine behind the bottom player bar.
# Ships its own platform backends (CoreAudio / WASAPI / ALSA / PulseAudio) and
# decoders (mp3 / flac / wav), so it adds no link dependencies of its own.
#
# miniaudio ships a CMakeLists.txt that builds its own node/graph test targets.
# SOURCE_SUBDIR points somewhere without one, so FetchContent only downloads the
# header and the implementation is compiled by our single player.cpp TU.
FetchContent_Declare(miniaudio
  GIT_REPOSITORY https://github.com/mackron/miniaudio.git
  GIT_TAG 0.11.25
  GIT_SHALLOW TRUE
  SOURCE_SUBDIR no-cmake-targets)
FetchContent_MakeAvailable(miniaudio)

# ---------------------------------------------------------------- stb_image
# Single-header, public-domain image decoding for the cover thumbnail; only the
# JPEG path is compiled (STBI_ONLY_JPEG in cover.cpp). stb has no release tags,
# so it is pinned to a commit, and as with miniaudio its own CMake is skipped.
FetchContent_Declare(stb
  GIT_REPOSITORY https://github.com/nothings/stb.git
  GIT_TAG 2c980bb59875b0d32144a71867fbdebb2f77cd20
  SOURCE_SUBDIR no-cmake-targets)
FetchContent_MakeAvailable(stb)

# ---------------------------------------------------------------- bundled fonts
# Three Open Font License faces are linked into the binary so Chinese / Korean /
# Russian / accented-Latin text in paths and titles renders without depending on
# any system font:
#   [0] primary  - Noto Sans SC  (CJK incl. Extension A + Latin)
#   [1] fallback - Noto Sans     (Latin-Ext / Greek / Cyrillic / combining marks)
#   [2] fallback - Noto Sans KR  (Hangul + Hanja)
# They are merged into one ImFont with ImFontConfig::MergeMode.
#
# The fonts are embedded IN FULL - no pyftsubset step - so no script can be
# silently dropped. The only transformation is instancing the upstream variable
# fonts to a static Regular (see assets/fonts/instance.sh).
# Blobs are gzip-compressed and inflated once at startup.
set(RENCM_FONTS
  "NotoSansSC-Regular.ttf"  # [0] primary
  "NotoSans-Regular.ttf"    # [1] fallback
  "NotoSansKR-Regular.ttf"  # [2] fallback
)

# Compressing the blobs needs both zlib (to inflate at startup) and the external
# gzip tool (to compress at build time). Windows has no gzip by default and may
# lack zlib, so this stays optional and falls back to embedding the fonts as-is
# (16.6 MB instead of 9.2 MB). RENCM_FONT_COMPRESSED=OFF forces the fallback.
option(RENCM_FONT_COMPRESSED "gzip-compress the embedded fonts" ON)
find_package(ZLIB QUIET)
find_program(GZIP_EXE NAMES gzip)

set(_gz 0)
if(RENCM_FONT_COMPRESSED AND ZLIB_FOUND AND GZIP_EXE)
  set(_gz 1)
else()
  message(STATUS "fonts embedded uncompressed (zlib=${ZLIB_FOUND}, gzip=${GZIP_EXE})")
endif()

# How the blobs are linked in differs by toolchain: GCC/Clang assemble a GNU .S
# with .incbin, MSVC cannot, so there they become Win32 RCDATA resources that
# embedded_font.cpp reads back with FindResource.
if(MSVC)
  set(_rc "/* generated by CMake from gui/CMakeLists.txt - do not edit */\n")
else()
  set(_asm "/* generated by CMake from gui/CMakeLists.txt - do not edit */\n")
  if(APPLE)
    string(APPEND _asm "    .section __TEXT,__const\n")
  else()
    string(APPEND _asm "    .section .rodata\n")
  endif()
endif()

set(_idx 0)
foreach(_name IN LISTS RENCM_FONTS)
  set(_src "${CMAKE_CURRENT_SOURCE_DIR}/assets/fonts/${_name}")
  if(NOT EXISTS "${_src}")
    message(FATAL_ERROR "bundled font missing: ${_src}\nSee assets/fonts/README.md")
  endif()

  set(_blob "${_src}")
  if(_gz)
    set(_blob "${CMAKE_CURRENT_BINARY_DIR}/rencm_font${_idx}.ttf.gz")
    execute_process(COMMAND "${GZIP_EXE}" -9 -n -c "${_src}" OUTPUT_FILE "${_blob}"
      RESULT_VARIABLE _gz_rc)
    if(NOT _gz_rc EQUAL 0 OR NOT EXISTS "${_blob}")
      message(FATAL_ERROR "failed to gzip ${_name} (${GZIP_EXE})")
    endif()
  endif()

  if(MSVC)
    file(TO_NATIVE_PATH "${_blob}" _blob_native)
    string(APPEND _rc "\"rencm_font${_idx}\" RCDATA \"${_blob_native}\"\n")
  else()
    string(APPEND _asm
      "    .globl rencm_font${_idx}_start\n"
      "rencm_font${_idx}_start:\n"
      "    .incbin \"${_blob}\"\n"
      "    .globl rencm_font${_idx}_end\n"
      "rencm_font${_idx}_end:\n")
  endif()
  math(EXPR _idx "${_idx}+1")
endforeach()

if(MSVC)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/embedded_font.rc" "${_rc}")
  set(RENCM_FONT_SRC "${CMAKE_CURRENT_BINARY_DIR}/embedded_font.rc")
else()
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/embedded_font.S" "${_asm}")
  set(RENCM_FONT_SRC "${CMAKE_CURRENT_BINARY_DIR}/embedded_font.S")
endif()

# ---------------------------------------------------------------- app
add_executable(rencm-gui
  src/main.cpp
  src/tasks.cpp
  src/jobs.cpp
  src/headless.cpp
  src/ui.cpp
  src/browser.cpp
  src/playerbar.cpp
  src/theme.cpp
  src/player.cpp
  src/cover.cpp
  src/embedded_font.cpp
  ${RENCM_FONT_SRC})
target_include_directories(rencm-gui PRIVATE src ${miniaudio_SOURCE_DIR} ${stb_SOURCE_DIR})
target_compile_definitions(rencm-gui PRIVATE GL_SILENCE_DEPRECATION)
target_link_libraries(rencm-gui PRIVATE rencm::rencm imgui_backend)
target_compile_definitions(rencm-gui PRIVATE RENCM_FONT_COMPRESSED=${_gz})
if(_gz)
  target_link_libraries(rencm-gui PRIVATE ZLIB::ZLIB)
endif()

if(APPLE)
  target_link_libraries(rencm-gui PRIVATE
    "-framework OpenGL"
    "-framework Cocoa"
    "-framework IOKit"
    "-framework CoreVideo")
elseif(WIN32)
  # ImGui's OpenGL3 loader resolves GL entry points through opengl32. GLFW
  # brings its own user32/gdi32/shell32, and miniaudio loads ole32/advapi32
  # dynamically, so nothing else is required.
  target_link_libraries(rencm-gui PRIVATE opengl32)
elseif(UNIX)
  # miniaudio's ALSA / PulseAudio backend needs these on Linux.
  find_package(Threads REQUIRED)
  target_link_libraries(rencm-gui PRIVATE Threads::Threads ${CMAKE_DL_LIBS} m)
endif()
